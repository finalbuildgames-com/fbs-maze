cmake_minimum_required(VERSION 3.16)
project(FinalBuildMaze VERSION 0.1.0 LANGUAGES C)
include(GNUInstallDirs)
include(CTest)
option(FBS_BUILD_TESTS "Build module tests" ON)
option(FBS_BUILD_EXAMPLES "Build the example" ON)
add_library(fbs_maze src/maze.c)
add_library(fbs::maze ALIAS fbs_maze)
set_target_properties(fbs_maze PROPERTIES C_STANDARD 99 C_STANDARD_REQUIRED YES
  C_EXTENSIONS NO POSITION_INDEPENDENT_CODE ON WINDOWS_EXPORT_ALL_SYMBOLS ON EXPORT_NAME maze)
target_include_directories(fbs_maze PUBLIC
  $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
  $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>)
if(NOT MSVC)
  target_link_libraries(fbs_maze PUBLIC m)
  target_compile_options(fbs_maze PRIVATE -Wall -Wextra -Wpedantic -ffp-contract=off)
endif()
if(FBS_BUILD_TESTS AND BUILD_TESTING)
  add_executable(fbs_test_maze tests/test_maze.c)
  set_target_properties(fbs_test_maze PROPERTIES C_STANDARD 99 C_STANDARD_REQUIRED YES C_EXTENSIONS NO)
  target_link_libraries(fbs_test_maze PRIVATE fbs::maze)
  if(NOT MSVC)
    target_compile_options(fbs_test_maze PRIVATE -UNDEBUG -ffp-contract=off)
  endif()
  add_test(NAME maze COMMAND fbs_test_maze)
  set_tests_properties(maze PROPERTIES WORKING_DIRECTORY ${CMAKE_CURRENT_SOURCE_DIR}/tests)
endif()
if(FBS_BUILD_EXAMPLES)
  add_executable(fbs_maze_example examples/basic.c)
  set_target_properties(fbs_maze_example PROPERTIES C_STANDARD 99 C_STANDARD_REQUIRED YES C_EXTENSIONS NO)
  target_link_libraries(fbs_maze_example PRIVATE fbs::maze)
  if(BUILD_TESTING)
    add_test(NAME maze_example COMMAND fbs_maze_example)
  endif()
endif()
install(TARGETS fbs_maze EXPORT FinalBuildMazeTargets)
install(DIRECTORY include/ DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
install(EXPORT FinalBuildMazeTargets NAMESPACE fbs::
  DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/FinalBuildMaze)
install(FILES LICENSE THIRD_PARTY_NOTICES DESTINATION ${CMAKE_INSTALL_DATADIR}/licenses/fbs-maze)
install(DIRECTORY third_party/ DESTINATION ${CMAKE_INSTALL_DATADIR}/licenses/fbs-maze/upstream OPTIONAL)
